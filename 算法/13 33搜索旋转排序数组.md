- `finMinIndex`方法里的else在该题并不需要, 如果数组中有重复元素, 才需要判断相等的场景, 此题数组里没有相同的值
- ==记得所有的`mid`中值的计算要放到`while`循环里面==
```
class Solution {
    public int search(int[] nums, int target) {
        // 先找到数组里的最小值, 那么就找到了两段不同的升序序列[0, minIndex-1], [minIndex, nums.length-1]
        int minIndex = finMinIndex(nums);
        int left = 0, right = nums.length - 1;
        // 记下来就开始判断target在哪一段升序序列里面, 然后对那一段升序序列进行二分查找
        if(nums[right] < target){
            // 说明位于第一段
            return searchBinary(nums, target, left, minIndex-1);
        }else if(nums[right] > target){
            return searchBinary(nums, target, minIndex, right);
        }else{
            return minIndex;
        }
    }

    private int searchBinary(int[] nums, int target, int left, int right){
        // 开始循环
        while(left <= right){
	        // 首先找到中值
	        int mid = (left + right) >>> 1;
            if(nums[mid] < target){
                left = mid + 1;
            }else if(nums[mid] > target){
                right = mid - 1;
            }else{
                return mid;
            }
        }
        return -1;
    }

    private int finMinIndex(int[] nums){
        int left = 0, right = nums.length - 1;
        while(left < right){
	        int mid = (left + right) >>> 1;
            if(nums[mid] > nums[right]){
                // 此时说明mid位于第一段升序序列
                left = mid + 1;
            }else if(nums[mid] < nums[right]){
                // 之所以不等于mid-1是因为, 此时的mid一定在第二段升序序列
                // 如果mid在第一段, 那么一定nums[mid] > nums[right], 所有此时mid一定在第二段
                // 而最小值是第二段升序序列的开头, 所以最小值一定在mid或者mid左侧, 所以令right=mid
                right = mid;
            }else{
                right--;
            }
        }
        return right;
    }
}
```
