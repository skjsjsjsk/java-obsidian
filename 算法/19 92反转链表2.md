class Solution {
    public ListNode reverseBetween(ListNode head, int left, int right) {
        // 创建一个哨兵节点, 它指向head, 值为0(防止left=1时做反转)
        ListNode dummyHead = new ListNode(0, head);
        // 找到left所指向节点的前一个节点
        ListNode preLeftNode = dummyHead;
        for (int i=0; i<left-1; i++){
            preLeftNode = preLeftNode.next;
        }
        // 定义需要进行反转部分链表的头部
        ListNode leftNode = preLeftNode.next;

        ListNode pre = null;
        ListNode cur = leftNode;

        // 开始反转
        for(int i=0; i<right-left+1; i++){
            ListNode tem = cur.next;
            cur.next = pre;
            pre = cur;
            cur = tem;
        }

        // 连接到原链表中  pre是反转后的链表的头部, cur是尾部
        // 反转为链表后, pre是头部(原链表中right位置的节点), leftNode是尾部(原链表中left位置的节点), cur是反转区间后面的第一个节点
        preLeftNode.next = pre;
        leftNode.next = cur;

        return dummyHead.next;
    }
}
